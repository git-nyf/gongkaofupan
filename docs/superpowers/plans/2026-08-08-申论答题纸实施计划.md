# 申论答题纸实施计划

> **供智能体执行：** 必须使用 `subagent-driven-development` 按任务推进；每个功能先写失败测试，再做最小实现。并行任务只修改各自独占文件，主智能体负责共享入口、整合与最终验证。

**目标：** 在现有公考记忆卡中交付独立“申论”页面，支持严格 25 格排版、粘贴灌格、字数模板、每 200 字水印、超限警示、普通备注、复盘链接、选中文字批注、连接箭头、快捷键、本地草稿和可点击链接的打印 PDF。

**架构：** 纯逻辑层负责排版、草稿和链接解析，React 页面层负责剪贴板、选择、批注和快捷键，独立 SVG 连接层负责箭头几何，复盘页只增加哈希定位适配。页面数据保存在带版本号的 `localStorage` 草稿中，不新增服务端接口或数据库结构。

**技术栈：** React 18、TypeScript、React Router、Lucide React、Vitest、Testing Library、原生 `Intl.Segmenter`、Selection/Range、SVG、CSS 打印媒体查询。

describe('申论答题纸排版', () => {

  it('每行固定 25 格并把英数两个字符装入一格', () => {
    const result = layoutShenlunText('甲，AB12乙', 200);
    expect(SHENLUN_COLUMNS).toBe(25);
    expect(result.usedCells).toBe(5);
    expect(result.rows[0].cells.slice(0, 5).map((cell) => cell?.text)).toEqual([
      '甲', '，', 'AB', '12', '乙',
    ]);
  });

  it('不让闭合标点单独出现在行首', () => {
    const result = layoutShenlunText(`${'甲'.repeat(25)}，`, 200, { segmentWords: false });
    expect(result.rows[0].cells.filter(Boolean)).toHaveLength(24);
    expect(result.rows[1].cells.slice(0, 2).map((cell) => cell?.text)).toEqual(['甲', '，']);
  });

  it('按回车强制换行但不把空白计入字数', () => {
    const result = layoutShenlunText('甲 乙\n丙', 200);
    expect(result.usedCells).toBe(3);
    expect(result.rows[1].cells[0]?.text).toBe('丙');
  });

  it('到达 200 字时生成左侧水印位置', () => {
    const result = layoutShenlunText('甲'.repeat(200), 400, { segmentWords: false });
    expect(result.markers).toEqual([{ count: 200, rowIndex: 7 }]);

  it('超出模板后追加警示行并只标红超限格', () => {
    const result = layoutShenlunText('甲'.repeat(202), 200, { segmentWords: false });
    expect(result.rows).toHaveLength(9);
    expect(result.overflowCells).toBe(2);
    expect(result.rows[8].cells.slice(0, 2).every((cell) => cell?.overflow)).toBe(true);
  });
});

const [hashTargetId, setHashTargetId] = useState<string | null>(null);

useEffect(() => {
  if (loading) return;
  const match = /^#review-item-([A-Za-z0-9_-]+)$/.exec(window.location.hash);
  const target = match ? items.find(({ id }) => id === match[1]) : undefined;
  if (!target) return;
  const board = boards.find(({ sections }) => sections.some(({ id }) => id === target.sectionId));
  if (!board) return;
  setCollapsedBoardIds((current) => {
    const next = new Set(current);
    next.delete(board.id);
    return next;
  });
  setSelectedByBoard((current) => ({ ...current, [board.id]: target.sectionId }));
  setActiveImage(target.sectionId, target.id);
  setHashTargetId(target.id);
}, [boards, items, loading]);

useEffect(() => {
  if (!hashTargetId) return;
  const frame = window.requestAnimationFrame(() => {
    document.getElementById(`review-item-${hashTargetId}`)?.scrollIntoView({
      behavior: 'smooth',
      block: 'center',
    });
  });
  return () => window.cancelAnimationFrame(frame);
}, [activeBySection, hashTargetId, selectedByBoard]);
